// Mints a brand-new skos:Concept in the given scheme. The full definition // (skos:inScheme, skos:prefLabel) lives in ONTOLOGY_GRAPH, matching every // existing concept there. But every pre-existing concept ALSO has a // lightweight mirror asserted directly in HOLONS_GRAPH -- just // "a skos:Concept ; rdfs:label "..."@en", no skos:inScheme/skos:prefLabel // -- because SHACL validation on this dataset (push_turtle's shapes_graph) // is scoped to the holons+shacl graphs, not unioned with the ontology // graph, so an sh:class skos:Concept check can only ever see that mirror. // Without it, a value using a freshly-minted concept fails validation even // though the concept is "real" in the ontology graph. (Confirmed against // Euro/Human/English/PoundSterling, all of which carry exactly this // mirror shape.) Both graphs are written in one INSERT DATA so a concept // is never left half-created. function buildCreateConceptSparql(iri, label, scheme){ return [ "PREFIX skos: ", "PREFIX rdfs: ", "INSERT DATA {", " GRAPH <" + ONTOLOGY_GRAPH + "> {", " <" + iri + "> a skos:Concept ;", " skos:inScheme <" + scheme + "> ;", " skos:prefLabel " + langLiteral(label) + " .", " }", " GRAPH <" + HOLONS_GRAPH + "> {", " <" + iri + "> a skos:Concept ;", " rdfs:label " + langLiteral(label) + " .", " }", "}" ].join("\n"); } // sh:in is a closed list -- minting the concept above isn't enough to // make it a valid value, since SHACL checks membership in the literal // RDF list, not "any skos:Concept in this scheme". This splices the new // IRI onto the end of *every* sh:in list that shares this path, across // every NodeShape that declares one (solar-system#bodyType, for example, // has two separate property shapes with two separate lists -- a new body // type needs appending to both, or it would validate for one target // class but not the other). Same last-node-swap technique used to add // geo:District to PlaceKindExclusivityShape's sh:xone: find each list's // terminal node (?lastNode rdf:rest rdf:nil) and replace that rdf:nil // with a fresh one-element tail. SPARQL Update mints one fresh blank node // per WHERE solution, so this is safe even when several distinct lists // match at once. function buildAppendConceptToInSparql(path, newIri){ return [ "PREFIX sh: ", "PREFIX rdf: ", "DELETE {", " GRAPH <" + SHACL_GRAPH + "> { ?lastNode rdf:rest rdf:nil }", "}", "INSERT {", " GRAPH <" + SHACL_GRAPH + "> {", " ?lastNode rdf:rest _:newNode .", " _:newNode rdf:first <" + newIri + "> .", " _:newNode rdf:rest rdf:nil .", " }", "}", "WHERE {", " GRAPH <" + SHACL_GRAPH + "> {", " ?shape sh:property ?prop .", " ?prop sh:path <" + path + "> ; sh:in ?listHead .", " ?listHead rdf:rest* ?lastNode .", " ?lastNode rdf:rest rdf:nil .", " }", "}" ].join("\n"); }
